## Quick Reference

### Using the Enhanced RecipeFlashcard Component

```typescript

<RecipeFlashcard
  recipe={currentRecipe}
```

### AI Service - Expanding Recipes

```typescript
import { aiService } from '@/services/ai.service';

// Convert a suggested recipe to full recipe with details
const suggestedRecipe: SuggestedRecipe = {
  title: "Chicken Stir Fry",
  description: "Quick and healthy",
  difficulty: "beginner",
  total_time_minutes: 25,
  servings: 4,
  match_score: 92,
  ingredients_you_have: ["chicken", "broccoli"],
  ingredients_you_need: ["soy sauce", "ginger"],
  preview_steps: ["Cut chicken", "Stir fry vegetables"],
  cuisine_type: "Asian",
};

// Expand to full recipe
const fullRecipe = await aiService.expandRecipeFromSuggestion(suggestedRecipe);

// fullRecipe now has:
// - Full ingredients list with amounts/units
// - Detailed numbered steps
// - Nutrition estimates
// - Cooking tips
```

## Data Structures

### FlashcardRecipe (UI Display)

```typescript
interface FlashcardRecipe {
  id: string;                           // 'ai-1', 'ai-2', etc.
  name: string;                         // Display title
  image: string;                        // Image URL from Unsplash
  prepTime: string;                     // "25 min" or "1h 30m"
  difficulty: 'Easy' | 'Medium' | 'Hard';
  matchPercentage: number;              // 0-100
  isChefChoice: boolean;                // First recipe = true
  ingredients: FlashcardIngredient[];   // Preview list
  originalRecipe?: SuggestedRecipe;     // Full AI data for expansion
}
```

### SuggestedRecipe (AI Output)

```typescript
interface SuggestedRecipe {
  title: string;
  description: string;
  cuisine_type: string;
  difficulty: RecipeDifficulty;         // 'beginner' | 'intermediate' | 'advanced'
  total_time_minutes: number;
  servings: number;
  match_score: number;                  // 0-100
  ingredients_you_have: string[];       // From fridge scan
  ingredients_you_need: string[];       // To buy
  preview_steps: string[];              // Brief overview
  meal_type?: MealType;                 // 'breakfast' | 'lunch' | 'dinner'
}
```

### ExtractedRecipe (Full Recipe)

```typescript
interface ExtractedRecipe {
  title: string;
  description: string;
  cuisine_type?: string;
  difficulty: RecipeDifficulty;
  prep_time_minutes?: number;
  cook_time_minutes?: number;
  total_time_minutes: number;
  active_time_minutes: number;
  servings: number;
  ingredients: Ingredient[];            // With amounts and units
  steps: RecipeStep[];                  // Numbered instructions
  tools?: string[];                     // Required equipment
  nutrition_estimate?: {
    calories?: number;
    protein_g?: number;
    carbs_g?: number;
    fat_g?: number;
  };
  tips?: string[];                      // Helpful cooking tips
}
```

### Ingredient (Full Detail)

```typescript
interface Ingredient {
  name: string;                         // "chicken breast"
  amount?: number;                      // 1.5
  unit?: IngredientUnit;                // "lb"
  category: IngredientCategory;         // "meat"
  notes?: string;                       // "boneless, skinless"
  is_optional?: boolean;                // false
}
```

### RecipeStep (Cooking Instructions)

```typescript
interface RecipeStep {
  step_number: number;                  // 1, 2, 3...
  instruction: string;                  // "Heat oil in pan over medium-high heat"
  duration_minutes?: number;            // 5
  temperature?: string;                 // "350°F"
  image_url?: string;                   // Optional step image
}
```

## Event Flow

### Save to Favorites Flow

```
User taps heart icon
    ↓
RecipeFlashcard.onFavoritePress(recipe)
    ↓
recipe-results.handleFavoritePress(recipe)
    ↓
aiService.expandRecipeFromSuggestion(recipe.originalRecipe)
    ↓
AI generates full recipe (~2-3 seconds)
    ↓
recipeStore.addRecipe(fullRecipe)
    ↓
Firebase saves to 'recipes' collection
    ↓
Success alert: "Saved! Recipe added to your collection"
    ↓
UI updates: Heart icon turns solid red
```

### View Detail Flow

```
User taps recipe card
    ↓
RecipeFlashcard.onCardPress(recipe)
    ↓
recipe-results.handleCardPress(recipe)
    ↓
router.push('/suggested-recipe-detail', { recipe, imageUrl })
    ↓
SuggestedRecipeDetailScreen loads
    ↓
aiService.expandRecipeFromSuggestion(recipe)
    ↓
Display full recipe with:
  - Ingredients (grouped by have/need)
  - Detailed cooking steps
  - Nutrition information
    ↓
User can save or start cooking
```

## Performance Tips

### 1. Cache Expanded Recipes

```typescript
const recipeCache = new Map<string, ExtractedRecipe>();

async function getExpandedRecipe(suggested: SuggestedRecipe) {
  const cacheKey = suggested.title;

  if (recipeCache.has(cacheKey)) {
    return recipeCache.get(cacheKey)!;
  }

  const expanded = await aiService.expandRecipeFromSuggestion(suggested);
  recipeCache.set(cacheKey, expanded);
  return expanded;
}
```

### 2. Preload Next Recipe

```typescript
// In recipe-results.tsx
useEffect(() => {
  // Preload next recipe's expansion when user views current one
  const nextRecipe = currentRecipes[currentIndex + 1];
  if (nextRecipe?.originalRecipe) {
    aiService.expandRecipeFromSuggestion(nextRecipe.originalRecipe)
      .catch(() => {}); // Silent preload
  }
}, [currentIndex]);
```

### 3. Background Save

```typescript
const handleFavoritePress = async (recipe: FlashcardRecipe) => {
  // Optimistic UI update
  setIsSaved(true);

  // Background save
  try {
    const fullRecipe = await aiService.expandRecipeFromSuggestion(recipe.originalRecipe);
    await addRecipe(fullRecipe);
  } catch (error) {
    // Revert on error
    setIsSaved(false);
    Alert.alert('Error', 'Failed to save recipe');
  }
};
```

## Common Issues

### Issue: Heart button triggers detail view

**Solution:** Add `e.stopPropagation()` and `pointerEvents="box-only"`

```typescript
<Pressable
  onPress={(e) => {
    e.stopPropagation(); // Critical!
    handleFavorite();
  }}
  pointerEvents="box-only"
/>
```

### Issue: AI expansion takes too long

**Solution:** Show loading state and implement timeout

```typescript
const [isExpanding, setIsExpanding] = useState(false);

const expandWithTimeout = async (recipe: SuggestedRecipe) => {
  setIsExpanding(true);
  const timeout = setTimeout(() => {
    throw new Error('Request timeout');
  }, 15000); // 15 second timeout

  try {
    const expanded = await aiService.expandRecipeFromSuggestion(recipe);
    clearTimeout(timeout);
    return expanded;
  } finally {
    setIsExpanding(false);
  }
};
```

### Issue: Duplicate saves

**Solution:** Debounce save button

```typescript
const [isSaving, setIsSaving] = useState(false);

const handleSave = async () => {
  if (isSaving) return; // Prevent duplicate saves

  setIsSaving(true);
  try {
    await addRecipe(fullRecipe);
  } finally {
    setIsSaving(false);
  }
};
```
